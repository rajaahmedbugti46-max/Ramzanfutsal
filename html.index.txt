<script>
const sheetID = "PASTE_YOUR_SHEET_ID_HERE";

function showSection(id) {
    document.querySelectorAll('.section').forEach(sec => sec.classList.remove('active'));
    document.getElementById(id).classList.add('active');
}

function loadTable(sheetName, elementID) {
    fetch(https://opensheet.elk.sh/${sheetID}/${sheetName})
    .then(res => res.json())
    .then(data => {
        let table = "<table><tr>";
        Object.keys(data[0]).forEach(key => {
            table += <th>${key}</th>;
        });
        table += "</tr>";

        data.forEach(row => {
            table += "<tr>";
            Object.values(row).forEach(cell => {
                table += <td>${cell}</td>;
            });
            table += "</tr>";
        });

        table += "</table>";
        document.getElementById(elementID).innerHTML = table;
    });
}

// Load Group Tables
loadTable("Group A", "groupA");
loadTable("Group B", "groupB");
loadTable("Group C", "groupC");
loadTable("Group D", "groupD");

// Load Top Scorers
loadTable("Top Scorer Tracker", "scorers");

// Load Knockout Bracket
fetch(https://opensheet.elk.sh/${sheetID}/Knockout Stage)
.then(res => res.json())
.then(data => {
    let html = "<h2>Knockout Bracket</h2>";
    data.forEach(row => {
        html += <div>${Object.values(row).join(" ")}</div>;
    });
    document.getElementById("knockout").innerHTML = html;
});

// Show default tab
showSection('groupA');
